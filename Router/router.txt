# router.txt: the router script the mod runs for every order (brief 23). Written in the router language
# (src\RouterScript\README.md); not Python.
#
# One entry function, route(order). It asks Haiku to turn the player's order into sentences (lines), checks
# every sentence, and asks the mod to plan each one. Nothing is applied here: the mod shows the planned
# lines to the player, who says yes or no.
#
# The host functions it calls (contract: briefs\23_contract.md section 2):
#   vocabulary()          the VOCABULARY block: one "name: a, b, c" line per kind
#   pending()             none, or {"order": text, "question": text} left by an earlier ASK
#   haiku(input)          Haiku's answer to the prompt (mod\Router\haiku_prompt.md) and this input
#   plan(form, slots)     {"ok": true, "line": text} or {"ok": false, "reason": text}
#   query(form, slots)    {"ok": true, "text": text} (the answer; lines separated by a new line) or
#                         {"ok": false, "reason": text}. Brief 26: for questions; changes nothing.
#   log(text)             one line in the mod's log
#
# route() gives back one of:
#   {"result": "plans",  "lines": [text, ...]}      every sentence is planned; the player confirms
#   {"result": "ask",    "question": text}          Haiku asked; NOTHING is planned or handed on
#   {"result": "cannot", "reason": text}            Haiku said it can not; nothing is planned
#   {"result": "answer", "lines": [text, ...]}      every sentence was a question and was answered
#   {"result": "failed", "reasons": [text, ...]}    a sentence did not pass a check, or plan said no,
#                                                   or questions and changes were mixed
#
# The rules in the shared block below (sections 1 to 8) are the same text as in router_lab\parse_results.txt,
# the lab's results parser, and the check router_lab\check_router_sync.ps1 keeps them that way. Change them
# in both files. Only section 9, the driver, differs.
#
# Top-level lines run again on every run, and a function can not assign a global (README section 6): so the
# top level here only builds tables and forms (the shared block), and route() keeps what it learns in local
# names and in the dicts MEMORY and VOCAB, which it clears or fills at the start of every call.

# ==== shared rules: begin ====

# ---- 1. Lowercase ----
# Everything is compared in lowercase from here on, so nothing is rejected for its capitals.
# (Read-back sentences are lowercase too.)
# Brief 27: a question that ends with "?" is read as if it ended with "." ("is bob married?"); "? " in the
# middle of a line is a period too. ASK: and CANNOT: lines are left as they are (their question keeps its "?").

function to_lowercase(line):
    s = lower(line)
    if starts_with(trim(s), "ask:") or starts_with(trim(s), "cannot:"):
        return s
    s = replace(s, "? ", ". ")
    s = trim(s)
    if ends_with(s, "?"):
        s = replace_last_mark(s)
    return s

# The text without its last character, then a period.
function replace_last_mark(s):
    out = ""
    i = 0
    while i < len(s) - 1:
        out = out + s[i]
        i = i + 1
    return out + "."

# ---- 2. Wording ----
# Words that mean the same. Add a row to a table to add a word; nothing else needs to change.

# Words that carry no meaning; dropped.
FILLER = ["a", "an", "the"]

# Where a command starts. A verb word is only changed there, so "make weapon group ..." and
# "clear weapon group ..." keep their "group". Section 3 splits at the same places.
SEPARATORS = [". ", " and then ", ", and then ", "\n "]

# Verb words at the start of a command -> the one verb the forms use.
# Brief 21 (user 2026-09-28): build intent is only ever "build" or "construct".
VERBS = {"select": "select", "group": "select", "gather": "select", "construct": "build"}
# User 2026-10-03: the sentence forms stay as they are; new words for the same thing are rows here. Verb words
# for the sentences the agent-player build added ("retreat" = move is added next to its forms, section 5).
VERBS["pull back"] = "move"
VERBS["fall back"] = "move"
VERBS["put out"] = "extinguish"
VERBS["douse"] = "extinguish"
VERBS["patch up"] = "tend"
VERBS["treat"] = "tend"
VERBS["heal"] = "tend"
VERBS["shoot"] = "attack"

# Phrases anywhere in the line -> the one phrase the forms use. Longer phrases first.
PHRASES = [
    ["have melee weapons equipped", "have melee weapons"],
    ["melee weapons equipped", "have melee weapons"],
    ["all of my pawns", "everyone"],
    ["all my pawns", "everyone"],
    ["wield", "have"],
    ["closest", "nearest"],
    ["map center", "map centre"]
]

function without_filler(line):
    kept = []
    for w in split(line, " "):
        if not (w in FILLER):
            append(kept, w)
    return join(kept, " ")

# In a select-pawns command, "that who", "that", or no word before "have" is read as "who"
# (user 2026-10-02: the form is "select PAWNS who have CONDITION."; "that" is not part of it).
# The line is cut into commands here only to find the ones that start with "select"; section 3 cuts it
# the same way.
function select_who(line):
    s = replace(line, ", and then ", ". ")
    s = replace(s, " and then ", ". ")
    out = []
    for part in split(s, ". "):
        m = find_pattern("^select (.+?)(?: that who| that| who)? have (.+)$", part)
        if m != none:
            part = "select " + m[1] + " who have " + m[2]
        append(out, part)
    return join(out, ". ")

function wording(line):
    # "\n " marks the start of the line, so the first command's verb is found like the others.
    s = "\n " + line + " "
    for p in PHRASES:
        for after in [" ", ".", ","]:
            s = replace(s, " " + p[0] + after, " " + p[1] + after)
    for verb in keys(VERBS):
        for sep in SEPARATORS:
            s = replace(s, sep + verb + " ", sep + VERBS[verb] + " ")
    s = replace(s, "\n ", "")
    return select_who(without_filler(trim(s)))

# ---- 3. Commands in one line ----
# A period followed by a space, or "and then", ends one command and starts the next. "and" alone never
# does (user 2026-10-01); a plain "and" before an action verb is refused in read_sentence().

function commands(line):
    s = replace(line, ", and then ", ". ")
    s = replace(s, " and then ", ". ")
    out = []
    for part in split(s, ". "):
        if trim(part) != "":
            append(out, trim(part))
    return out

# The first word of a line or a template, in lowercase (backticks ignored). A template can only match
# a line that starts with the same word, so the other forms are not tried.
function first_word(text):
    words = split(trim(replace(lower(text), "`", "")), " ")
    return words[0]

FIXED = {}

# lists: the slots that hold several names ("a, b and c"); they are shown as lists.
# readback: the template filled back in for the player (the form's own template when it is the same).
# place: for the storage forms, the PLACE template; remember() fills it to remember the storage.
# start: the first word of the template.
function add_form(name, template, readback, lists, place):
    append(forms, {"name": name, "template": template, "readback": readback, "lists": lists, "place": place, "start": first_word(template)})

function form(name, template, lists):
    add_form(name, template, template, lists, "")

function new_form(name, template, lists):
    NEW_NAMES[name] = true
    form(name, template, lists)

function fixed_form(name, template, lists, fixed):
    FIXED[template] = fixed
    new_form(name, template, lists)

# One form for each PLACE: before and after are the words around it.
function storage_form(name, before, after, lists):
    for place in PLACES:
        add_form(name, before + place + after, before + place + after, lists, place)

# A storage form that says "those" (brief 20): the parse template has "those" written out, the read-back
# has {OBJECTS}, which resolve() fills with the remembered objects.
function those_form(name, before, after, lists):
    add_form(name, before + "those" + after, before + "{OBJECTS}" + after, lists, "")

function help_rows(questions, word):
    rows = []
    for h in HELP:
        if is_query(h["name"]) == questions:
            if word == "" or first_word(h["line"]) == lower(word):
                append(rows, h)
    return rows

# The answer of "show questions." / "show orders." (name = the form, slots = its slots).
function help_answer(name, slots):
    questions = name == "query questions"
    kind = "order"
    if questions:
        kind = "question"
    word = ""
    if has(slots, "WORD"):
        word = trim(slots["WORD"])
    rows = help_rows(questions, word)
    if len(rows) == 0:
        return ["No " + kind + " sentence starts with '" + word + "'."]
    out = []
    if word == "" and len(rows) > HELP_CAP:
        words = []
        counts = {}
        for h in rows:
            w = first_word(h["line"])
            if not has(counts, w):
                append(words, w)
                counts[w] = 0
            counts[w] = counts[w] + 1
        for w in words:
            if counts[w] == 1:
                append(out, w + ": 1 sentence")
            else:
                append(out, w + ": " + text(counts[w]) + " sentences")
        append(out, "Say 'show " + kind + "s for WORD.' for the sentences that start with WORD.")
        return out
    for h in rows:
        append(out, h["line"] + " = " + h["what"])
    return out

function is_help(name):
    return name == "query questions" or name == "query orders"

# Direction words (brief 18): north = up = top, south = down = bottom, west = left, east = right, and
# the corners ("northeast", "top right", "north-east"). North is +z, east is +x (Verse/Rot4.cs:155-158).
DIRECTION_WORDS = {
    "north": "north", "up": "north", "top": "north", "upper": "north",
    "south": "south", "down": "south", "bottom": "south", "lower": "south",
    "west": "west", "left": "west",
    "east": "east", "right": "east"
}
CORNERS = ["northeast", "northwest", "southeast", "southwest"]

# The one direction name ("top right" -> "northeast"), or none.
function direction_name(words):
    parts = split(replace(lower(trim(words)), "-", " "), " ")
    north_south = ""
    east_west = ""
    for part in parts:
        if part == "":
            continue
        if part in CORNERS and len(parts) == 1:
            return part
        if not has(DIRECTION_WORDS, part):
            return none
        d = DIRECTION_WORDS[part]
        if d == "north" or d == "south":
            if north_south != "":
                return none
            north_south = d
        else:
            if east_west != "":
                return none
            east_west = d
    if north_south == "" and east_west == "":
        return none
    return north_south + east_west

function whole_number_between(s, low, high):
    n = number(s)
    if n == none:
        return false
    return n == n - n % 1 and n >= low and n <= high

# The vocabulary the order was written against: "name: a, b, c" lines, as in vocabulary.txt, kept in
# lowercase. VOCAB["rooms"] = ["kitchen", ...]; the recipes line "recipes: BENCH: a, b; BENCH: c" is kept as
# VOCAB["recipes of BENCH"]. Empty = no vocabulary was given (the lab with an old results file): the checks
# against it are skipped.
VOCAB = {}

function names_of(text):
    out = []
    for item in split(lower(text), ","):
        if trim(item) != "":
            append(out, trim(item))
    return out

# text: the VOCABULARY block, one line per kind.
function read_vocabulary(text):
    for line in split(text, "\n"):
        m = find_pattern("^([^:]+): (.*)$", trim(line))
        if m == none:
            continue
        kind = trim(lower(m[1]))
        if kind == "letters":
            # "letters: wanderer joins (accept, reject); ransom for Bob (pay, refuse)"
            labels = []
            for part in split(lower(m[2]), "; "):
                r = find_pattern("^(.*?) \\((.*)\\)$", trim(part))
                if r != none:
                    append(labels, trim(r[1]))
                    VOCAB["options of " + trim(r[1])] = names_of(r[2])
                elif trim(part) != "":
                    append(labels, trim(part))
            VOCAB["letters"] = labels
        elif kind == "operations":
            # "operations: Bob: install bionic arm, amputate left leg; Dana: ..."
            for part in split(m[2], "; "):
                r = find_pattern("^([^:]+): (.*)$", trim(part))
                if r != none:
                    VOCAB["operations of " + trim(lower(r[1]))] = names_of(r[2])
        elif kind == "recipes":
            for part in split(m[2], "; "):
                r = find_pattern("^([^:]+): (.*)$", trim(part))
                if r != none:
                    VOCAB["recipes of " + trim(lower(r[1]))] = names_of(r[2])
        else:
            VOCAB[kind] = names_of(m[2])

# Is item (lowercase) on the vocabulary line `kind`?
function listed(kind, item):
    if len(keys(VOCAB)) == 0:
        return true
    return has(VOCAB, kind) and item in VOCAB[kind]

function listed_any(kinds, item):
    for kind in kinds:
        if listed(kind, item):
            return true
    return false

# {"text": the one written form, "why": ""} or {"why": what is wrong}.
function read_spot(raw):
    s = trim(raw)
    # "112, 87 (6 tiles north of bob)": the brackets are how an answer described the cell; dropped.
    m = find_pattern("^(.+?) *\\(.*\\)$", s)
    if m != none:
        s = trim(m[1])
    m = find_pattern("^near (.+)$", s)
    if m != none:
        s = trim(m[1])
    m = find_pattern("^\\(? *([0-9]+) *, *([0-9]+) *\\)?$", s)
    if m != none:
        return {"text": text(number(m[1])) + ", " + text(number(m[2])), "why": ""}
    m = find_pattern("^([a-z0-9]+) tiles? (.+?) of (.+)$", s)
    if m != none:
        n = m[1]
        if has(NUMBER_WORDS, n):
            n = text(NUMBER_WORDS[n])
        if not whole_number_between(n, 1, 300):
            return {"why": "the distance must be a whole number of tiles, 1 to 300"}
        d = direction_name(m[2])
        if d == none:
            return {"why": "'" + m[2] + "' is not a direction: north, south, east, west or a corner"}
        base = read_spot_name(m[3])
        if base["why"] != "":
            return base
        return {"text": text(number(n)) + " tiles " + d + " of " + base["text"], "why": ""}
    m = find_pattern("^(.+?) of (.+)$", s)
    if m != none and direction_name(m[1]) != none:
        return {"why": "say how many tiles: 'N tiles " + direction_name(m[1]) + " of " + m[2] + "'"}
    return read_spot_name(s)

function read_spot_name(s):
    s = replace(trim(s), "map center", "map centre")
    if s in ["those", "them", "they", "it", "there", "here"]:
        return {"why": "'" + s + "' is not a place; " + SPOT_SHAPES}
    if find_pattern("^[0-9]", s) != none:
        return {"why": "'" + s + "' is not a place; a cell is two numbers, x then z: '112, 87'"}
    if listed_any(SPOT_LINES, s):
        return {"text": s, "why": ""}
    return {"why": "'" + s + "' is not a place on the map; " + SPOT_SHAPES}

function read_area(raw):
    s = trim(raw)
    m = find_pattern("^\\(? *([0-9]+) *, *([0-9]+) *\\)? *to *\\(? *([0-9]+) *, *([0-9]+) *\\)?$", s)
    if m != none:
        return {"text": text(number(m[1])) + ", " + text(number(m[2])) + " to " + text(number(m[3])) + ", " + text(number(m[4])), "why": ""}
    m = find_pattern("^([0-9]+) *(?:by|x) *([0-9]+) (?:around|at|near|centred on|centered on) (.+)$", s)
    if m != none:
        if not whole_number_between(m[1], 1, 250) or not whole_number_between(m[2], 1, 250):
            return {"why": "an area's size must be two whole numbers, 1 to 250"}
        spot = read_spot(m[3])
        if spot["why"] != "":
            return spot
        return {"text": text(number(m[1])) + " by " + text(number(m[2])) + " around " + spot["text"], "why": ""}
    if find_pattern("^[0-9]+ *(?:by|x) *[0-9]+$", s) != none:
        return {"why": "say where the area is: 'W by H around SPOT'"}
    return {"why": "'" + s + "' is not an area; " + AREA_SHAPES}

# "" when every SPOT / AREA slot (and LAYER, THING, NAME of agent 03's questions) is fine, else what is wrong.
# The SPOT / AREA slots are rewritten in their one form.
function check_place(name, slots):
    for key in keys(slots):
        r = none
        if starts_with(key, "SPOT"):
            r = read_spot(slots[key])
        elif starts_with(key, "AREA"):
            r = read_area(slots[key])
        if r != none:
            if r["why"] != "":
                return r["why"]
            slots[key] = r["text"]
    if has(slots, "LAYER"):
        if not has(MAP_LAYERS, slots["LAYER"]):
            return "LAYER must be terrain, resources, buildings, roof or creatures"
        slots["LAYER"] = MAP_LAYERS[slots["LAYER"]]
    if name == "query where":
        if starts_with(slots["NAME"], "closest "):
            return "write 'where is nearest THING', not 'closest'"
        if starts_with(slots["NAME"], "nearest ") or slots["NAME"] in ["those", "them", "they", "it"]:
            return "NAME must be a colonist, prisoner, animal, landmark or zone"
    if has(slots, "THING") and (slots["THING"] in ["those", "them", "they", "it", "thing", "things"]):
        return "name the thing to look for"
    if starts_with(name, "query list"):
        for item in slots["THINGS"]:
            if item in ["those", "them", "they", "it", "thing", "things"]:
                return "name the things to look for"
    return ""

# The numbers in a SPOT / AREA form (as check_place wrote it) that the player must have said: coordinates,
# a distance in tiles, an area's size. Never the numbers inside a name.
function place_numbers(s):
    out = []
    m = find_pattern("^([0-9]+), ([0-9]+) to ([0-9]+), ([0-9]+)$", s)
    if m != none:
        return [number(m[1]), number(m[2]), number(m[3]), number(m[4])]
    m = find_pattern("^([0-9]+) by ([0-9]+) around (.+)$", s)
    if m != none:
        append(out, number(m[1]))
        append(out, number(m[2]))
        s = m[3]
    m = find_pattern("^([0-9]+), ([0-9]+)$", s)
    if m != none:
        append(out, number(m[1]))
        append(out, number(m[2]))
        return out
    m = find_pattern("^([0-9]+) tiles ", s)
    if m != none:
        append(out, number(m[1]))
    return out

# "" when every such number was said, else which slot holds one that was not.
function place_made_up(slots, said):
    for key in keys(slots):
        if starts_with(key, "SPOT") or starts_with(key, "AREA"):
            for n in place_numbers(slots[key]):
                if not (n in said):
                    return key + " " + slots[key]
    return ""

# FACING (which way a building is turned) and SIDE (which wall of a new room gets the door): one of the
# four sides; "on the south side" is read as "south".
SIDES = ["north", "south", "east", "west"]
NOT_A_BUILDING = ["those", "them", "they", "it", "room", "floor", "blueprint"]

function side_name(words):
    s = trim(words)
    m = find_pattern("^(?:on )?(.+?)(?: side)?$", s)
    if m != none:
        s = trim(m[1])
    d = direction_name(s)
    if d == none or not (d in SIDES):
        return none
    return d

# "" when fine. Runs after check_place (agent 03), so SPOT and AREA are already in their one form.
function check_build_place(name, slots):
    for key in ["FACING", "SIDE"]:
        if has(slots, key):
            d = side_name(slots[key])
            if d == none:
                return key + " must be north, south, east or west"
            slots[key] = d
    if has(slots, "BUILDING"):
        b = slots["BUILDING"]
        first = split(b, " ")[0]
        if b in NOT_A_BUILDING or first in ["room", "floor"]:
            return "'" + b + "' is not a building: a room is 'build room AREA', a floor is 'build floor FLOOR in area AREA'"
        if not listed("buildings", b):
            return "'" + b + "' is not on the buildings line (what the colony can build now)"
    if has(slots, "FLOOR") and not listed("floors", slots["FLOOR"]):
        return "'" + slots["FLOOR"] + "' is not on the floors line (the floors the colony can lay now)"
    if name == "build near":
        # A bare name that is a building or a bench is an ANCHOR: the older "near ANCHOR" form takes it (it
        # finds the standing building and places next to it). With no vocabulary every name goes there.
        spot = slots["SPOT"]
        if find_pattern("^[0-9]", spot) == none:
            if len(keys(VOCAB)) == 0 or listed_any(["benches", "buildings", "storage buildings"], spot):
                return "'" + spot + "' is a building; 'near ANCHOR' is the older build form"
    return ""

# The older form "build BUILDING [made of STUFF]." (no room, no anchor) is the last one tried, and a sentence
# no form takes is reported with the last form's problem. When the sentence was one of the forms above, this
# gives that form's problem instead ("FACING must be ..."), so the player reads the real reason. "" = none.
function build_place_why(slots):
    line = "build " + slots["BUILDING"]
    if has(slots, "STUFF"):
        line = line + " made of " + slots["STUFF"]
    line = line + "."
    for f in forms:
        if not has(BUILD_PLACE, f["name"]):
            continue
        s = parse(f["template"], line)
        if s == none:
            continue
        why = check_place(f["name"], s)
        if why == "":
            why = check_build_place(f["name"], s)
        if why != "":
            return why
    # "build campfire north of bob": the older side form with a colonist or landmark as the building.
    s = parse("build {BUILDING} {DIRECTION} of {ANCHOR}.", line)
    if s != none and direction_name(s["DIRECTION"]) != none:
        return anchor_not_place(s)
    return ""

# The older forms' ANCHOR is a building that stands on the map. A colonist, landmark or zone there is a
# SPOT said without "near" or "N tiles": "" when fine, else what to write instead. Skipped with no vocabulary.
function anchor_not_place(slots):
    if not has(slots, "ANCHOR") or len(keys(VOCAB)) == 0:
        return ""
    a = slots["ANCHOR"]
    if listed_any(["colonists", "landmarks", "zones"], a) and not listed_any(["benches", "buildings", "storage buildings"], a):
        return "'" + a + "' is not a building; write 'near " + a + "' or 'at N tiles DIRECTION of " + a + "'"
    return ""

# The entry of the "hostiles" vocabulary line that is t (as "raider 1") or t + " (down)", else none. With no
# vocabulary at all (the lab with an old results file) every name is accepted as it is.
function hostile_entry(t):
    if len(keys(VOCAB)) == 0:
        return t
    if not has(VOCAB, "hostiles"):
        return none
    for e in VOCAB["hostiles"]:
        if e != "none" and (e == t or e == t + DOWN):
            return e
    return none

# True when the vocabulary is empty, or the hostiles line has an entry other than "none" that is not down.
function enemies_in_view():
    if len(keys(VOCAB)) == 0:
        return true
    if not has(VOCAB, "hostiles"):
        return false
    for e in VOCAB["hostiles"]:
        if e != "none" and not ends_with(e, DOWN):
            return true
    return false

# "" when the pawn-action form's slots are fine, else the reason. The TARGET slot is rewritten without a
# "(down)" the player may have copied from the vocabulary line ("raider 2 (down)" -> "raider 2").
function check_pawn_actions(name, slots):
    if not (name in PAWN_ACTION_FORMS):
        return ""
    if has(slots, "DOER"):
        if slots["DOER"] in ["them", "they", "those"]:
            return "name who does it, not '" + slots["DOER"] + "'"
        if not listed("colonists", slots["DOER"]):
            return "'" + slots["DOER"] + "' is not a colonist, so it can not be the one who does it"
        if name in ["rescue", "arrest"] and has(slots, "PAWN") and slots["DOER"] == slots["PAWN"]:
            return slots["PAWN"] + " can not " + name + " themselves; name another colonist"
    if has(slots, "TARGET"):
        t = slots["TARGET"]
        m = find_pattern("^(.+?) \\(down\\)$", t)
        if m != none:
            t = trim(m[1])
        if t == "nearest enemy" and name != "attack":
            return "'nearest enemy' is only for attack; " + name + " needs the name of one enemy"
        if t in ["them", "they", "those", "it"]:
            return "'" + t + "' is not an enemy; name the enemy, like 'raider 1'"
        if len(keys(VOCAB)) > 0:
            if t == "nearest enemy":
                if not enemies_in_view():
                    return NO_ENEMY
            else:
                entry = hostile_entry(t)
                if entry == none:
                    if not enemies_in_view():
                        if name == "attack":
                            return NO_ENEMY
                    return "'" + t + "' is not an enemy in view"
                if name == "attack" and ends_with(entry, DOWN):
                    return "'" + t + "' is already down; capture them instead"
                if name == "capture" and not ends_with(entry, DOWN):
                    return "'" + t + "' is not down; only a downed enemy can be captured"
        slots["TARGET"] = t
    if name == "prioritize" and not (slots["WORK"] in WORKS):
        return "WORK must be one of " + join_list(WORKS)
    for key in ["WEAPON", "APPAREL", "ITEM"]:
        if has(slots, key):
            why = check_item_words(name, key, slots)
            if why != "":
                return why
    return ""

# The item of equip / wear / drop / pick up is a plain name. When the form with a place or with a COUNT did not
# fit, the next form (without them) would take the whole rest as the item name ("rifle at nowhere", "two herbal
# medicine"); those are refused here with the reason of the part that was meant.
function check_item_words(name, key, slots):
    v = slots[key]
    first = split(v, " ")[0]
    m = find_pattern("^(.+?) (?:at|near) (.+)$", v)
    if m != none:
        r = read_spot(m[2])
        if r["why"] != "":
            return r["why"]
    if find_pattern("^[0-9]", v) != none or (name == "pick up" and has(NUMBER_WORDS, first) and not (first in ["first", "second", "third", "fourth", "noon", "midnight"])):
        if name == "pick up" and not has(slots, "COUNT"):
            return "COUNT must be a whole number, written in digits"
        return "'" + v + "' must be a name, not a number of things"
    if m != none:
        return "'" + v + "' must be a name; put the place last, like 'at 10, 12'"
    return ""

# "is mary jane married": PAWN takes the shortest text, so a name of several words ends up half in RELATION.

function fix_pawn_name(name, slots):

        return none

# "" when fine: the value of a drug group entry must fit its field.
function check_drug_entry(slots):
    if not has(DRUG_FIELDS, slots["FIELD"]):
        return "FIELD must be " + join_list(keys(DRUG_FIELDS))
    if not listed("drug groups", slots["NAME"]):
        return "'" + slots["NAME"] + "' is not a drug group in this colony"
    if not listed("drugs", slots["DRUG"]):
        return "'" + slots["DRUG"] + "' is not a drug in this colony's drug groups"
    kind = DRUG_FIELDS[slots["FIELD"]]

    if kind == "flag":

        return "VALUE for '" + slots["FIELD"] + "' must be a number, 0 or more"
    elif kind == "percent" and v > 100:
        return "VALUE for '" + slots["FIELD"] + "' is a percent, 0 to 100"
    elif kind == "count" and v != v - v % 1:
        return "VALUE for '" + slots["FIELD"] + "' must be a whole number"
    return ""

# Is the operation (recipe, part) one of those listed for that pawn? Skipped when the pawn has no list.

# The checks of the forms of brief 23: numbers, then names against the vocabulary, then the form's own.
function check_new(name, slots):

# Slots that hold a number (the mod gets them as numbers).
NUMBER_SLOTS = ["COUNT", "N", "SLOT", "DISTANCE", "OPTION", "MIN", "MAX", "DEGREES", "FROM", "TO", "VALUE", "HOURS", "DAYS", "DAY"]

# The slots with the number slots as numbers.
function with_numbers(slots):
    out = {}
    for key in slots:
        if key in NUMBER_SLOTS and number(slots[key]) != none:
            out[key] = number(slots[key])
        else:
            out[key] = slots[key]
    return out

# Every number in the order: digit runs ("10pm" gives 10) and number words.
function numbers_in(order):
    found = []
    s = lower(order) + " "
    digits = ""
    word = ""
    i = 0
    while i < len(s):
        c = s[i]
        if contains("0123456789", c):
            digits = digits + c
        elif digits != "":
            append(found, number(digits))
            digits = ""
        if contains("abcdefghijklmnopqrstuvwxyz", c):
            word = word + c
        elif word != "":
            if has(NUMBER_WORDS, word):
                append(found, NUMBER_WORDS[word])
            word = ""
        i = i + 1
    return found

# An hour counts as said when the player said it, or said it on a 12-hour clock (22 from "10pm",
# 0 from "12" or "midnight").
function hour_said(h, said):
    return h in said or (h >= 12 and h - 12 in said) or (h == 0 and 12 in said)

# "" when every number in the line was in the order (and the answer to a question), else which one wasn't.
# DEGREES may be below zero; the minus sign is not a number the player has to say twice.
function made_up(slots, said):
    p = place_made_up(slots, said)
    if p != "":
        return p
    for key in ["COUNT", "N", "SLOT", "DISTANCE", "OPTION", "MIN", "MAX", "HOURS", "DAYS", "DAY"]:
        # OPTION is also the text of a letter's choice (brief 27): only a number has to have been said.
        if key == "OPTION" and has(slots, key) and number(slots[key]) == none:
            continue
        if has(slots, key) and not (number(slots[key]) in said):
            return key + " " + slots[key]
    # A drug group entry's VALUE is true / false or a number; the number must be said.
    if has(slots, "VALUE") and number(slots["VALUE"]) != none and not (number(slots["VALUE"]) in said):
        return "VALUE " + slots["VALUE"]
    if has(slots, "DEGREES"):
        d = number(slots["DEGREES"])
        if d < 0:
            d = -d
        if not (d in said):
            return "DEGREES " + slots["DEGREES"]
    for key in ["FROM", "TO"]:
        if has(slots, key) and not hour_said(number(slots[key]), said):
            return key + " " + slots[key]
    return ""

# Memory for one player order (one "## n." block of the results file; one call of route()): it lasts across
# lines, periods and "and then", and is cleared at the next order.
#   them / they = the pawns selected earlier in the order. Never objects.
#   those       = the objects selected earlier in the order. Never pawns.
#   those       = the objects an earlier sentence in the order acted on (brief 20): the storage of a storage
#                 sentence (its PLACE as written), or the building of a "build" sentence with where it
#                 goes (brief 21: "build option" and "build room" set nothing), or the TARGETS of a
#                 building-control sentence (brief 23). Never a room, never pawns. A later sentence may say
#                 "those" where it names the objects it acts on: the "(those)" forms of section 4 and the
#                 TARGETS of the building-control forms.
# MEMORY["pawns"] is set by the form "select pawns"; MEMORY["objects"] by the storage, build and
# building-control forms (see remember()). The latest sentence replaces what was remembered, but a
# sentence that only says "those" keeps what it pointed at. "at" counts the sentences of the order (the
# first is 0); pawns_at / objects_at are the positions of the sentences that set them: the mod is given
# that position as THEM / THOSE instead of names (contract, brief 23 section 2).
# Priority (brief 22/23): only storage has a priority or an item filter. "those" from a storage sentence is
# storage; "those" from a "build" sentence is storage only when the BUILDING is on the vocabulary line
# "storage buildings" ("build 2 shelf in storeroom. set those to important priority" is fine, "build 2 bed
# in barracks. set those to important priority" is refused).
MEMORY = {"pawns": [], "objects": [], "pawns_at": -1, "objects_at": -1, "kind": "", "buildings": [], "at": -1}

function clear_memory():
    MEMORY["pawns"] = []
    MEMORY["objects"] = []
    MEMORY["pawns_at"] = -1
    MEMORY["objects_at"] = -1
    MEMORY["kind"] = ""
    MEMORY["buildings"] = []
    MEMORY["at"] = -1

# Called once for every sentence of the order, so "at" is its position.
function begin_sentence():
    MEMORY["at"] = MEMORY["at"] + 1

function refused(why):
    return {"problem": why, "names": {}, "refs": {}}

# Replaces them / they / those in the slots with what was remembered. Gives a dict:
#   "problem"  "" when fine, else what is wrong
#   "names"    the slots with them / they / those written out as names (the read-back and the lab use it)
#   "refs"     the slots with them / those left out and THEM / THOSE = the position of the earlier sentence
#              (what the mod is given). The "(those)" forms get their {OBJECTS} in "names" here.
function resolve(f, slots):
    name = f["name"]
    takes_those = ends_with(name, "(those)")
    if (has(slots, "ROOM") and slots["ROOM"] == "those") or (has(slots, "DOOR") and slots["DOOR"] == "those") or (has(slots, "ROOMS") and "those" in slots["ROOMS"]) or (name == "move" and slots["NAME"] == "those"):
        return refused("'those' means objects, never a room; name the room")
    if has(slots, "ANCHOR") and slots["ANCHOR"] in ["those", "them", "they"]:
        return refused("'" + slots["ANCHOR"] + "' can not be what a building goes next to; name the building")
    used_them = false
    pawn_names = []
    pawn_refs = []
    if has(slots, "PAWNS"):
        for item in slots["PAWNS"]:
            if item in ["them", "they"]:
                if len(MEMORY["pawns"]) == 0:
                    return refused("'" + item + "' has no pawns selected earlier in this order")
                used_them = true
                for p in MEMORY["pawns"]:
                    append(pawn_names, p)
            elif item == "those":
                return refused("'those' means objects, not pawns; say 'them'")
            else:
                append(pawn_names, item)
                append(pawn_refs, item)
    one_pawn = ""
    if has(slots, "PAWN") and slots["PAWN"] in ["them", "they"]:
        if len(MEMORY["pawns"]) != 1:
            return refused("'" + slots["PAWN"] + "' needs exactly one pawn selected earlier in this order")
        used_them = true
        one_pawn = MEMORY["pawns"][0]
    if has(slots, "THINGS"):
        for item in slots["THINGS"]:
            if item in ["them", "they"]:
                return refused("'" + item + "' means pawns, not things; name the things")
            elif item == "those":
                return refused("'those' means the storage or buildings an earlier sentence acted on, not things to put in")
    used_those = false
    those_targets = false
    if takes_those:
        if len(MEMORY["objects"]) == 0:
            return refused("'those' has no storage or buildings from an earlier sentence in this order")
        if MEMORY["kind"] == "control":
            return refused("'those' are buildings with controls, not storage; storage is the only thing with a priority or an item list")
        if MEMORY["kind"] == "build":
            for b in MEMORY["buildings"]:
                if not listed("storage buildings", b):
                    return refused("'" + b + "' is not storage, so 'those' can not take a priority or an item list; only stockpile zones and " + join_list(get(VOCAB, "storage buildings", [])) + " can")
        used_those = true
    if has(slots, "TARGETS"):
        for item in slots["TARGETS"]:
            if item in ["them", "they"]:
                return refused("'" + item + "' means pawns, not buildings; name the buildings")
        if "those" in slots["TARGETS"]:
            if len(slots["TARGETS"]) != 1:
                return refused("'those' must stand alone; name the other buildings in another sentence")
            if has(slots, "ROOM"):
                return refused("'those' already means the buildings; leave the room out")
            if len(MEMORY["objects"]) == 0:
                return refused("'those' has no storage or buildings from an earlier sentence in this order")
            if MEMORY["kind"] == "storage":
                return refused("'those' are storage, not something this sentence can act on")
            used_those = true
            those_targets = true
    names = {}
    refs = {}
    for key in slots:
        if key == "PAWNS":
            names[key] = pawn_names
            if len(pawn_refs) > 0:
                refs[key] = pawn_refs
        elif key == "PAWN" and one_pawn != "":
            names[key] = one_pawn
        elif key == "TARGETS" and those_targets:
            names[key] = MEMORY["objects"]
        else:
            names[key] = slots[key]
            refs[key] = slots[key]
    if takes_those:
        names["OBJECTS"] = MEMORY["objects"]
    if used_them:
        refs["THEM"] = MEMORY["pawns_at"]
    if used_those:
        refs["THOSE"] = MEMORY["objects_at"]
    return {"problem": "", "names": names, "refs": refs}

# After a form matched (slots = its names): what it selected is remembered for the rest of the order.
function remember(f, slots, refs):
    name = f["name"]
    if name == "select pawns":
        MEMORY["pawns"] = slots["PAWNS"]
        MEMORY["pawns_at"] = MEMORY["at"]
    elif f["place"] != "":
        MEMORY["objects"] = [fill(f["place"], slots)]
        MEMORY["objects_at"] = MEMORY["at"]
        MEMORY["kind"] = "storage"
        MEMORY["buildings"] = []
    elif name == "build":
        what = slots["BUILDING"]
        if has(slots, "ANCHOR"):
            if has(slots, "DIRECTION"):
                what = what + " " + slots["DIRECTION"] + " of " + slots["ANCHOR"]
            else:
                what = what + " near " + slots["ANCHOR"]
        if has(slots, "ROOM"):
            what = what + " in " + slots["ROOM"]
        MEMORY["objects"] = [what]
        MEMORY["objects_at"] = MEMORY["at"]
        MEMORY["kind"] = "build"
        MEMORY["buildings"] = [slots["BUILDING"]]
    elif has(TARGET_RULES, name) and not has(refs, "THOSE"):
        objects = []
        for t in slots["TARGETS"]:
            if has(slots, "ROOM"):
                append(objects, t + " in " + slots["ROOM"])
            else:
                append(objects, t)
        MEMORY["objects"] = objects
        MEMORY["objects_at"] = MEMORY["at"]
        MEMORY["kind"] = "control"
        MEMORY["buildings"] = []

# Sections 1 to 7 applied to one sentence (a command): the same for the lab and for the mod.

function is_note(line):
    low = lower(trim(line))
    return starts_with(low, "ask:") or starts_with(low, "cannot:") or is_bare_note(line)

# Agent 10 (item 31): "CANNOT" or "ASK" alone (no colon, no reason, a period allowed) is a note too, with
# an empty reason. Haiku wrote a bare "CANNOT" in run h1 and it did not parse.
function is_bare_note(line):
    low = lower(trim(line))
    return low == "cannot" or low == "cannot." or low == "ask" or low == "ask."

# A note as the driver reads it: a bare one gets its colon ("CANNOT" -> "cannot:").
function note_of(line):
    if is_bare_note(line):
        return replace(lower(trim(line)), ".", "") + ":"
    return trim(line)

# The first ASK: or CANNOT: line of an answer (a list of lines), as written, or none. An answer with one
# is stopped: nothing in it goes on (user 2026-10-02, brief 22).
function first_note(lines):
    for line in lines:
        if trim(line) != "" and is_note(line):
            return note_of(line)
    return none

# One answer line -> its commands (lowercase, through wording, cut at periods and "and then").
function sentences_of(line):
    return commands(wording(to_lowercase(line)))

# One command, already lowercase and through wording. Gives a dict with "kind":
#   "ok"         "form", "names" and "refs" (see resolve()), "readback" (what the player reads)
#   "and"        a plain "and" before an action verb
#   "made up"    "what": the number the player never said
#   "unresolved" "form", "why": them / those pointed at nothing, or at the wrong kind
#   "no match"   no form fits; "form" and "why" tell what it came closest to ("" when nothing did)
function read_sentence(line, said):
    begin_sentence()
    if not is_note(line) and find_pattern(AND_ACTION, line) != none:
        return {"kind": "and", "line": line}
    first = first_word(line)
    problem_form = ""
    problem = ""
    for f in forms:
        if f["start"] != first:
            continue
        slots = parse(f["template"], line)
        if slots == none:
            continue
        # Slots the template itself decides (brief 27).
        if has(FIXED, f["template"]):
            for k in keys(FIXED[f["template"]]):
                slots[k] = FIXED[f["template"]][k]
        # List slots become lists, the way the router will use them.
        for key in slots:
            if key in f["lists"]:
                slots[key] = split_list(slots[key])
        wrong = check(f["name"], slots)
        if wrong == NOT_THIS_FORM:
            continue
        if wrong != "":
            problem_form = f["name"]
            problem = wrong
            continue
        invented = made_up(slots, said)
        if invented != "":
            return {"kind": "made up", "line": line, "what": invented}
        r = resolve(f, slots)
        if r["problem"] != "":
            return {"kind": "unresolved", "line": line, "form": f["name"], "why": r["problem"]}
        remember(f, r["names"], r["refs"])
        # The same template filled back in: what the player would read.
        return {"kind": "ok", "line": line, "form": f["name"], "names": r["names"], "refs": r["refs"], "readback": fill(f["readback"], r["names"])}
    return {"kind": "no match", "line": line, "form": problem_form, "why": problem}

# What the player reads when one sentence did not pass (r from read_sentence()).
function reason_of(r):
    line = r["line"]
    if r["kind"] == "and":
        return "\"" + line + "\": 'and' does not start a new command; use 'and then' or a period"
    if r["kind"] == "made up":
        return "\"" + line + "\": the number (" + r["what"] + ") was not in your order"
    if r["kind"] == "no match" and r["why"] == "":
        return "\"" + line + "\": I do not know how to do that"
    return "\"" + line + "\": " + r["why"]

# Haiku's answer as lines: no blank lines, no code fences.
function answer_lines(reply):
    out = []
    for line in split(reply, "\n"):
        t = trim(line)
        if t != "" and not starts_with(t, "```"):
            append(out, t)
    return out

function route(order):
    vocabulary_text = vocabulary()
    read_vocabulary(vocabulary_text)
    waiting = pending()
    original = order
    asked = ""
    answer = ""
    if waiting != none:
        # The second pass: the order from last time, Haiku's question, and what the player said now.
        original = waiting["order"]
        asked = waiting["question"]
        answer = order
    prompt_input = "VOCABULARY:\n" + trim(vocabulary_text) + "\nORDER:\n" + original
    if waiting != none:
        prompt_input = prompt_input + "\nASKED:\n" + asked + "\nANSWER:\n" + answer
    log("asking Haiku about the order: " + original)
    reply = haiku(prompt_input)
    if reply == none:
        reply = ""
    lines = answer_lines(reply)

    # An ASK or a CANNOT stops everything: no other line of the answer is looked at.
    note = first_note(lines)
    if note != none:
        m = find_pattern("^(ask|cannot): *(.*)$", note)
        if m != none and lower(m[1]) == "cannot":
            log("Haiku can not do it: " + m[2])
            # A bare CANNOT (agent 10, item 31) has an empty reason; the player still reads something.
            if trim(m[2]) == "":
                return {"result": "cannot", "reason": "Haiku said it cannot do that, without a reason."}
            return {"result": "cannot", "reason": m[2]}
        question = note
        if m != none:
            question = m[2]
        if trim(question) == "":
            question = "Could you say that again, with more detail?"
        log("Haiku asked: " + question)
        return {"result": "ask", "question": question}

    # Every sentence is read and checked; one bad sentence stops the whole order (agent 01: the tail of
    # route() is run_lines() below, shared with route_raw()).
    return run_lines(lines, numbers_in(original + " " + answer))

# ---- 9b. Raw sentences (agent inbox, agent_player brief 01): no Haiku call ----

# The same checks, questions and plans as route() does once Haiku has answered.
# lines: the router sentences, one per line. said: the numbers the player is allowed to use.
function run_lines(lines, said):
    # Every sentence is read and checked; one bad sentence stops the whole order.
    clear_memory()
    parsed = []
    reasons = []
    for line in lines:
        for command in sentences_of(line):
            r = read_sentence(command, said)
            if r["kind"] == "ok":
                append(parsed, r)
            else:
                append(reasons, reason_of(r))
    if len(parsed) == 0 and len(reasons) == 0:
        append(reasons, "Nothing in that order could be turned into a change.")
    # An order is all questions or all changes: a mix is refused as a whole, nothing is planned.
    formed = []
    for r in parsed:
        append(formed, r["form"])
    if mixed_problem(formed) != "":
        append(reasons, mixed_problem(formed))
    if len(reasons) > 0:
        log("the order failed: " + join(reasons, " | "))
        return {"result": "failed", "reasons": reasons}

    # Questions: the mod answers each one from the game's data; nothing is planned or changed.
    if is_query(parsed[0]["form"]):
        answered = []
        for r in parsed:
            # Agent 10 (item 5): show questions / show orders are answered by the script's own table.
            if is_help(r["form"]):
                for part in help_answer(r["form"], r["names"]):
                    append(answered, part)
                continue
            reply = query(r["form"], with_numbers(r["refs"]))
            if reply["ok"]:
                for part in split(reply["text"], "\n"):
                    if trim(part) != "":
                        append(answered, trim(part))
            else:
                append(reasons, reply["reason"])
        if len(reasons) > 0:
            log("the question failed: " + join(reasons, " | "))
            return {"result": "failed", "reasons": reasons}
        return {"result": "answer", "lines": answered}

    # Every sentence is planned, in order (them / those are positions of earlier sentences).
    planned_lines = []
    for r in parsed:
        planned = plan(r["form"], with_numbers(r["refs"]))
        if planned["ok"]:
            append(planned_lines, planned["line"])
        else:
            append(reasons, planned["reason"])
    if len(reasons) > 0:
        log("the order failed: " + join(reasons, " | "))
        return {"result": "failed", "reasons": reasons}
    return {"result": "plans", "lines": planned_lines}

# The text already holds router sentences (written by an agent): they go to the parser as they are.
# A line "ASK: ..." or "CANNOT: ..." is not special here: it is read like any sentence (the "ask" and "cannot"
# forms), and the game side refuses to plan it.
function route_raw(text):
    vocabulary_text = vocabulary()
    read_vocabulary(vocabulary_text)
    log("raw sentences, no Haiku call")
    lines = answer_lines(text)
    return run_lines(lines, numbers_in(text))
